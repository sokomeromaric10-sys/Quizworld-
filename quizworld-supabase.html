SQL — Tables QuizWorld 
-- TABLE 1 : Profils des joueurs
CREATE TABLE profiles (
  id            UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id       UUID REFERENCES auth.users ON DELETE CASCADE,
  username      TEXT UNIQUE NOT NULL,
  avatar        TEXT DEFAULT '👑',
  country       TEXT,
  total_score   INTEGER DEFAULT 0,
  games_played  INTEGER DEFAULT 0,
  best_score    INTEGER DEFAULT 0,
  correct_pct   NUMERIC(5,2) DEFAULT 0,
  created_at    TIMESTAMPTZ DEFAULT now()
);

-- TABLE 2 : Scores par partie
CREATE TABLE scores (
  id          UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id     UUID REFERENCES auth.users ON DELETE CASCADE,
  subject     TEXT NOT NULL,
  score       INTEGER NOT NULL,
  correct     INTEGER,
  total_q     INTEGER,
  difficulty  TEXT,
  played_at   TIMESTAMPTZ DEFAULT now()
);

-- SECURITE : Activer Row Level Security
ALTER TABLE profiles ENABLE ROW LEVEL SECURITY;
ALTER TABLE scores   ENABLE ROW LEVEL SECURITY;

-- Tout le monde peut voir les profils (pour le classement)
CREATE POLICY "Profils publics"
  ON profiles FOR SELECT USING (true);

-- Chaque joueur modifie uniquement son propre profil
CREATE POLICY "Modifier son profil"
  ON profiles FOR ALL
  USING (auth.uid() = user_id);

-- Tout le monde peut voir les scores (pour le classement)
CREATE POLICY "Scores publics"
  ON scores FOR SELECT USING (true);

-- Chaque joueur enregistre uniquement ses propres scores
CREATE POLICY "Enregistrer son score"
  ON scores FOR INSERT
  WITH CHECK (auth.uid() = user_id);
✅
Après "Run" vous verrez "Success. No rows returned" — c'est normal et correct !

// INSCRIPTION
async function inscrire(email, motDePasse, pseudo, pays, avatar) {
  const { data, error } = await supabase.auth.signUp({
    email: email, password: motDePasse
  })
  if (error) return { ok: false, message: error.message }
  await supabase.from('profiles').insert({
    user_id: data.user.id, username: pseudo,
    avatar: avatar, country: pays
  })
  return { ok: true, user: data.user }
}

// CONNEXION
async function connecter(email, motDePasse) {
  const { data, error } = await supabase.auth.signInWithPassword({
    email: email, password: motDePasse
  })
  if (error) return { ok: false, message: error.message }
  return { ok: true, user: data.user }
}

// DECONNEXION
async function deconnecter() {
  await supabase.auth.signOut()
}

// SAUVEGARDER UN SCORE
async function sauvegarderScore(sujet, score, correct, totalQ, difficulte) {
  const { data: { user } } = await supabase.auth.getUser()
  if (!user) return
  await supabase.from('scores').insert({
    user_id: user.id, subject: sujet, score: score,
    correct: correct, total_q: totalQ, difficulty: difficulte
  })
  const { data: profil } = await supabase
    .from('profiles').select('*').eq('user_id', user.id).single()
  await supabase.from('profiles').update({
    total_score:  profil.total_score + score,
    games_played: profil.games_played + 1,
    best_score:   Math.max(profil.best_score, score),
    correct_pct:  Math.round((correct / totalQ) * 100)
  }).eq('user_id', user.id)
}

// CLASSEMENT JOURNALIER
async function getClassementJour() {
  const hier = new Date()
  hier.setDate(hier.getDate() - 1)
  const { data } = await supabase
    .from('scores')
    .select('user_id, score, profiles(username, avatar, country)')
    .gte('played_at', hier.toISOString())
    .order('score', { ascending: false })
    .limit(10)
  return data
}
